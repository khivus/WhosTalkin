cmake_minimum_required(VERSION 3.21)
project(WhosTalkin VERSION 1.1.1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(WHOSTALKIN_BUILD_PLUGIN "Build the native TS3 plugin" ON)
option(WHOSTALKIN_BUILD_SIMULATOR "Build the standalone event simulator" ON)
option(WHOSTALKIN_BUILD_TESTS "Build automated tests" ON)
set(TS3_SDK_DIR "" CACHE PATH "Existing official ts3client-pluginsdk checkout (optional)")

add_library(speaker_core STATIC src/speaker_store.cpp src/person_filter.cpp)
target_include_directories(speaker_core PUBLIC src)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

if(WHOSTALKIN_BUILD_PLUGIN OR WHOSTALKIN_BUILD_SIMULATOR)
  # API 26 TS3 3.6.x ships Qt 5.15.2. A different Qt DLL must never be bundled.
  find_package(Qt5 5.15.2 EXACT REQUIRED COMPONENTS Core Gui Widgets)
  add_library(speaker_ui STATIC src/settings.cpp src/speaker_window.cpp src/controller.cpp
    src/controller_filters.cpp src/controller_window.cpp)
  target_include_directories(speaker_ui PUBLIC src)
  target_link_libraries(speaker_ui PUBLIC speaker_core Qt5::Widgets)
endif()

if(WHOSTALKIN_BUILD_PLUGIN)
  if(WIN32 AND (NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8))
    message(FATAL_ERROR "The Windows TS3 plugin requires an MSVC-compatible x64 toolchain and MSVC Qt 5.15.2.")
  endif()
  if(NOT TS3_SDK_DIR)
    include(FetchContent)
    FetchContent_Declare(ts3sdk
      GIT_REPOSITORY https://github.com/teamspeak/ts3client-pluginsdk.git
      GIT_TAG 4aa90a53aa150cbf81e13bc97e68c0431b26499f)
    FetchContent_MakeAvailable(ts3sdk)
    set(TS3_SDK_DIR "${ts3sdk_SOURCE_DIR}")
  endif()
  if(NOT EXISTS "${TS3_SDK_DIR}/include/ts3_functions.h")
    message(FATAL_ERROR "TS3_SDK_DIR must point to the official plugin SDK checkout.")
  endif()
  add_library(whostalkin SHARED src/plugin.cpp src/ts3_source.cpp)
  target_include_directories(whostalkin PRIVATE "${TS3_SDK_DIR}/include")
  target_link_libraries(whostalkin PRIVATE speaker_ui)
  target_compile_definitions(whostalkin PRIVATE WHOSTALKIN_VERSION="${PROJECT_VERSION}")
  set_target_properties(whostalkin PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
  if(WIN32)
    set_target_properties(whostalkin PROPERTIES OUTPUT_NAME whostalkin_win64)
  elseif(UNIX AND NOT APPLE)
    if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
      message(FATAL_ERROR "The Linux plugin target currently supports x86_64 only")
    endif()
    set_target_properties(whostalkin PROPERTIES OUTPUT_NAME whostalkin_linux_amd64
      BUILD_RPATH_USE_ORIGIN ON INSTALL_RPATH "$ORIGIN/..")
  else()
    message(FATAL_ERROR "Supported plugin platforms: Windows x64 and Linux x64")
  endif()
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Only x64 builds are supported")
  endif()
  install(TARGETS whostalkin DESTINATION plugins)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_custom_target(package_plugin
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/package.py"
        --binary "$<TARGET_FILE:whostalkin>" --version "${PROJECT_VERSION}"
        --output "${CMAKE_SOURCE_DIR}/dist"
      DEPENDS whostalkin VERBATIM)
  endif()
endif()

if(WHOSTALKIN_BUILD_SIMULATOR)
  add_executable(whostalkin_simulator src/simulator.cpp)
  target_link_libraries(whostalkin_simulator PRIVATE speaker_ui)
endif()

if(WHOSTALKIN_BUILD_TESTS)
  enable_testing()
  set(ui_test_environment "QT_QPA_PLATFORM=offscreen")
  if(WIN32)
    file(TO_CMAKE_PATH "$ENV{WINDIR}/Fonts" test_font_directory)
    list(APPEND ui_test_environment "QT_QPA_FONTDIR=${test_font_directory}")
  endif()
  add_executable(speaker_core_tests tests/speaker_store_tests.cpp)
  target_link_libraries(speaker_core_tests PRIVATE speaker_core)
  add_test(NAME speaker_core COMMAND speaker_core_tests)
  set_tests_properties(speaker_core PROPERTIES TIMEOUT 30)
  add_executable(person_filter_tests tests/person_filter_tests.cpp)
  target_link_libraries(person_filter_tests PRIVATE speaker_core)
  add_test(NAME person_filter COMMAND person_filter_tests)
  set_tests_properties(person_filter PROPERTIES TIMEOUT 30)
  if(TARGET speaker_ui)
    add_executable(speaker_ui_tests tests/ui_tests.cpp)
    target_link_libraries(speaker_ui_tests PRIVATE speaker_ui)
    add_test(NAME speaker_ui COMMAND speaker_ui_tests)
    set_tests_properties(speaker_ui PROPERTIES ENVIRONMENT "${ui_test_environment}" TIMEOUT 30)
    add_executable(feature_ui_tests tests/feature_ui_tests.cpp)
    target_link_libraries(feature_ui_tests PRIVATE speaker_ui)
    add_test(NAME feature_ui COMMAND feature_ui_tests)
    set_tests_properties(feature_ui PROPERTIES ENVIRONMENT "${ui_test_environment}" TIMEOUT 30)
  endif()
  if(TARGET whostalkin)
    add_executable(plugin_host_tests tests/plugin_host_tests.cpp)
    target_include_directories(plugin_host_tests PRIVATE "${TS3_SDK_DIR}/include")
    target_link_libraries(plugin_host_tests PRIVATE Qt5::Widgets)
    add_test(NAME plugin_host COMMAND plugin_host_tests "$<TARGET_FILE:whostalkin>")
    set_tests_properties(plugin_host PROPERTIES ENVIRONMENT "${ui_test_environment}" TIMEOUT 30)
  endif()
endif()
